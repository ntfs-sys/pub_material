Закон Био-Савара-Лапласа
$$d\vec{B} = \frac{\mu}{4pi} * \frac{I[d\vec{l} \times \vec r]}{r^3}$$

# 3.222

![[p07 222.png]]
$$\begin{aligned} &\text{Дано:} \\ &I = 1\text{ А} \\ &R = 0,1\text{ м} \\ &\text{а) } x = 0 \\ &\text{б) } x = 0,1\text{ м} \\ &B - ? \\ \\ &dB = \frac{\mu_0}{4\pi} \cdot \frac{I\,dl \cdot r}{r^3} = \frac{\mu_0}{4\pi} \cdot \frac{I\,dl}{r^2} \\ &dB_x = \frac{\mu_0 I\,dl}{4\pi r^2} \cos\alpha = \frac{\mu_0 I\,dl}{4\pi r^2} \cdot \frac{R}{r} = \frac{\mu_0 I\,dl R}{4\pi r^3} = \frac{\mu_0 I\,dl R}{4\pi (R^2 + x^2)^{3/2}} \\ &B = \int\limits_0^{2\pi R} \frac{\mu_0 I R}{4\pi (R^2 + x^2)^{3/2}} dl = \frac{\mu_0 I R^2}{2(R^2 + x^2)^{3/2}} \\ \\ &\text{а) } B = \frac{\mu_0 I}{2R} = \frac{4\pi \cdot 10^{-7} \cdot 1}{2 \cdot 0,1} = 2\pi \cdot 10^{-6}\text{ Тл} \approx 6,28\text{ мкТл} \\ \\ &\text{б) } B = \frac{4\pi \cdot 10^{-7} \cdot 1 \cdot 0,1^2}{2(0,1\sqrt{2})^3} = \frac{2\pi \cdot 10^{-9}}{10^{-3} (\sqrt{2})^3} = \frac{\pi}{\sqrt{2}} \cdot 10^{-6} \approx 2,22\text{ мкТл} \end{aligned}$$
# 3.224

![[p07 224.png]]
$$\begin{aligned} &dB = \frac{\mu_0}{4\pi} \cdot \frac{I\,dl \sin\alpha}{r^2} \\ &r = \frac{R}{\sin\alpha}, \quad dl = \frac{r\,d\alpha}{\sin\alpha} = \frac{R\,d\alpha}{\sin^2\alpha} \\ &dB = \frac{\mu_0}{4\pi} \frac{I\,d\alpha \sin\alpha}{R} \\ &B_1 = \int\limits_{\alpha_1}^{\alpha_2} \frac{\mu_0}{4\pi} \frac{I \sin\alpha\,d\alpha}{R_1} = \frac{\mu_0 I}{4\pi R_1} (-\cos\alpha)\Big\vert{}_{\alpha_1}^{\alpha_2} = \frac{\mu_0 I}{4\pi R_1} (\cos\alpha_1 - \cos\alpha_2) \\ &B_3 = B_1 = \frac{\mu_0 I}{4\pi R_1} \left(\cos\frac{\pi - \varphi}{2} - \cos\frac{\pi + \varphi}{2}\right) = \frac{\mu_0 I}{4\pi \left(\frac{d}{2}\cos\frac{\varphi}{2}\right)} \cdot 2\sin\frac{\varphi}{2} = \frac{\mu_0 I}{\pi d} \operatorname{tg}\frac{\varphi}{2} \\ &B_4 = B_2 = \frac{\mu_0 I}{\pi d} \operatorname{tg}\frac{\pi - \varphi}{2} = \frac{\mu_0 I}{\pi d} \operatorname{ctg}\frac{\varphi}{2} \\ &B = 2(B_1 + B_2) = \frac{2\mu_0 I}{\pi d} \left(\operatorname{ctg}\frac{\varphi}{2} + \operatorname{tg}\frac{\varphi}{2}\right) = \frac{4\mu_0 I}{\pi d \sin\varphi} \quad \text{(ошибка в } 2\text{ раза из-за } R_1 \text{, верно: } \frac{8\mu_0 I}{\pi d \sin\varphi}\text{)} \\ &B = \frac{8 \cdot 4\pi \cdot 10^{-7} \cdot 5}{\pi \cdot 0,16 \cdot 0,5} = 100\text{ мкТл} \end{aligned}$$
